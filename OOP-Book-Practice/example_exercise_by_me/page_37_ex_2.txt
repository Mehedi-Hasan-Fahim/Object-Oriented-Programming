#include<iostream>
#include<ctime>
using namespace std;
class stopwatch{
    clock_t begin,end;
    public:
       stopwatch();
       ~stopwatch();
       void start();
       void stop();
       void show();
};
stopwatch::stopwatch(){
    begin=end=0;
}
stopwatch::~stopwatch(){
    cout<<"Stopwatch object is destroyed"<<endl;
    
}
void stopwatch::start(){
    begin=clock()/CLOCKS_PER_SEC ;
}
void stopwatch::stop(){
    end=clock()/CLOCKS_PER_SEC ;
}
void stopwatch::show(){
    cout<<"Elapsed time "<<end-begin<<endl;
}

    stopwatch ob;

#include<iostream>
using namespace std;

class A {
    int x;
public:
    A(int n) {
        cout << "Constructing base class A" << endl;
        x = n;
    }
    ~A() {
        cout << "Destructing base class A" << endl;
    }

    void show_x() {
        cout << "x from A: " << x << endl;
    }
};

class B : public A {
    int y;
public:
    B(int n, int m) : A(n) {
        cout << "Constructing derived class B" << endl;
        y = m;
    }
    ~B() {
        cout << "Destructing derived class B" << endl;
    }

    void show_y() {
        cout << "y from B: " << y << endl;
    }
};

class C : public A {
    int z;
public:
    C(int n, int p) : A(n) {
        cout << "Constructing derived class C" << endl;
        z = p;
    }
    ~C() {
        cout << "Destructing derived class C" << endl;
    }

    void show_z() {
        cout << "z from C: " << z << endl;
    }
};

int main() {
    B objB(10, 20);
    C objC(30, 40);

    objB.show_x();//for class A in class B
    objB.show_y();//for class B

    objC.show_x();//for class A in class C
    objC.show_z();//for class C

    return 0;
}

#include<iostream>

class base{
  public:
  int a;
  int get(){
      return a;
  }
    
};
class derive:public base {
    /* copy of class base 
  
  public:
  int a;
  int get(){
      return a;
  }
    
    */
    
  public:
  int a;
  int get_d(){
      return a+get();
  }
    
};
int main(){
    base b;
    b.a=10;
    cout<<b.get()<<endl;

#include<iostream>
#include <cstdlib> // Include cstdlib for the exit() function
using namespace std;

#define Size 5

class stack {
    int top;
    char data[Size];
public:
    void init();
    void push();
    void pop();
    void display();
};

void stack::init() {
    top = -1;
}

void stack::push() {
    char item;
    if (top == Size - 1) {
        cout << "Stack is full";
    } else {
        cout << "Enter the item to push: ";
        cin >> item;
        top++;
        data[top] = item;
    }
}

void stack::pop() {
    if (top == -1) {
        cout << "Stack is Empty";
    } else {
        cout << "Popped item is " << data[top];
        top--;
    }
}

void stack::display() {
    if (top == -1) {
        cout << "Stack is Empty";
    } else {
        cout << "\nStack is...\n";
        for (int i = top; i >= 0; i--) {
            cout << data[i] << " ";
        }
    }
}

int main() {
    int ch;
    stack obj;
    obj.init(); // Initialize the stack before using it
    while (1) {
        cout << "\n1)Push\n";
        cout << "2)Pop\n";
        cout << "3)Display\n";
        cout << "4)Exit";
        cout << "\nEnter choice: ";
        cin >> ch;

        switch (ch) {
            case 1:
                obj.push();
                break;

            case 2:
                obj.pop();
                break;

            case 3:
                obj.display();
                break;

            case 4:
                exit(0);
            
            default:
                cout << "\nWrong Choice..";
        }
    }
}

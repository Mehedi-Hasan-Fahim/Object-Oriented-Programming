#include<iostream>
using namespace std;
class box{
    double l,w,h,v;
    public:
       box(double a,double b,double c);
       void display();
};
 box::box(double a,double b,double c){
       l=a;
       w=b;
       h=c;
       v=l*w*c;
 }
 void box::display(){
     cout<<"Volume is "<<v;
 }
 int main(){
     box ob(2.2 , 3.97 , 8.09);
     ob.display();
 }

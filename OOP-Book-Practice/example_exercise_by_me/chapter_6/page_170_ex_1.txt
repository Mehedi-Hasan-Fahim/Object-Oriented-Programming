#include<iostream>
#include<cstring>
using namespace std;

class strtype{
    char* p;
    int len;
    public:
    strtype(char *ptr){
        len=strlen(ptr);
        p=new char [len];
        if(!p){
            cout<<"Allocation error!"<<endl;
            exit(1);
        }
        strcpy(p,ptr);
    }
    
    char& operator[](int i){
        if(i<0 || i>=len){
            cout<<"Bounds error!!"<<endl;
            exit(1);
        }
        
        return p[i];
    }
    
};

int main(){
    strtype o1("Fahim");
    o1[4]='a';
    
    cout<<o1[4];
}

#include<iostream>
using namespace std;

class dynarray
{
int *p;
int size ;
public :
dynarray (int s); // pass size of array in s
int& put (int i); // return reference to element i
int get (int i); // return value of element i
dynarray operator=(dynarray& ob){
   
   if(size!=ob.size){
       cout << "Cannot copy arrays of different size !"<<endl;
       exit (1) ;
   }
   
   for(int i=0;i<size;i++){
       p[i] = ob.p[i] ;
   }
   
  return *this; 
    
}// create operator =() function
};

dynarray::dynarray(int s){
    p=new int [s] ;
    if(!p){
        cout<<"Allocation error!"<<endl;
        exit(1);
    }
    size = s ;
}

int& dynarray ::put(int i){
    if(i<0 || i >= size){
        cout<<"Bounds error!"<<endl;
        exit(1);
    }
    
    return p[i] ;
}

int dynarray ::get(int i){
    if(i<0 || i >= size){
        cout<<"Bounds error!"<<endl;
        exit(1);
    }
    
    return p[i] ;
}

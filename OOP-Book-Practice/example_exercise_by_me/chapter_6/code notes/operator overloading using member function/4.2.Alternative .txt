#include<iostream>
using namespace std;

class Number2;

class Number1 {
  int x, y;

public:
  Number1() {
    x = 0;
    y = 0;
  }

  Number1(int a, int b) {
    x = a;
    y = b;
  }

  void print() {
    cout << x << " " << y << endl;
  }
  
   int getX() const {
    return x;
  }

  int getY() const {
    return y;
  } 

  Number1 operator+(Number2& other);

};

class Number2 {
  int x, y;

public:
  Number2() {
    x = 0;
    y = 0;
  }

  Number2(int a, int b) {
    x = a;
    y = b;
  }

  void print() {
    cout << x << " " << y << endl;
  }

  int getX(){
    return x;
  }

  int getY(){
    return y;
  }
  
   Number2 operator+(Number1& other);
  
};

Number1 Number1::operator+(Number2& other){
  Number1 temp;
  temp.x = x + other.getX();
  temp.y = y + other.getY();
  return temp;
}

Number2 Number2::operator+(Number1& other){
  Number2 temp;
  temp.x = x + other.getX();
  temp.y = y + other.getY();
  return temp;
}

int main() {
  Number1 n1(1, 2);
  Number2 n2(3, 4);

  Number1 n3 = n1 + n2;
  Number2 n4 = n2 + n3;

  n1.print(); // Prints "1 2"
  n2.print(); // Prints "3 4"
  n3.print(); // Prints "4 6" (Number1 + Number2)
  n4.print();// Prints "7 10" (Number2 + Number3)
  
  return 0;
}

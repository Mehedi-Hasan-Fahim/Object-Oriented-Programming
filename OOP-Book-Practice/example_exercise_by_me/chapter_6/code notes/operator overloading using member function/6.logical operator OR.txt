#include <iostream>
using namespace std;

class Number{
  int x,y;
  
  public:
    Number(){
        x=0;
        y=0;
    }
    Number(int a,int b){
        x=a;
        y=b;
    }
    void print(){
        cout<<x<<" "<<y<<endl;
    }
    
    int operator || (Number ob){
        return (x||y) || (ob.x || ob.y);
    }

};    


int main()
{
    Number n1(1,2),n2(5,6),n3;
    
    if(n1||n2) cout<<"OK"<<endl;
    
    else cout<<"Error!"<<endl;
  
    return 0;
}

#include<iostream>
using namespace std;

class Number2; // Forward declaration of Number2 class

class Number1 {
  int x, y;

public:
  Number1() {
    x = 0;
    y = 0;
  }

  Number1(int a, int b) {
    x = a;
    y = b;
  }

  void print() {
    cout << x << " " << y << endl;
  }

  Number1 operator+(const Number1& other) const {
    Number1 temp;
    temp.x = x + other.x;
    temp.y = y + other.y;
    return temp;
  }

  // Conversion constructor to convert Number2 to Number1
  Number1(const Number2& other);

  // Getter functions for private members
  int getX() const {
    return x;
  }

  int getY() const {
    return y;
  }
};

class Number2 {
  int x, y;

public:
  Number2() {
    x = 0;
    y = 0;
  }

  Number2(int a, int b) {
    x = a;
    y = b;
  }

  void print() {
    cout << x << " " << y << endl;
  }

  // Conversion constructor to convert Number1 to Number2
  Number2(const Number1& other);

  // Getter functions for private members
  int getX() const {
    return x;
  }

  int getY() const {
    return y;
  }

  // Overload + operator for Number2 objects
  Number2 operator+(const Number2& other) const {
    Number2 temp;
    temp.x = x + other.x;
    temp.y = y + other.y;
    return temp;
  }
};

// Conversion constructor from Number2 to Number1
Number1::Number1(const Number2& other) {
  x = other.getX();
  y = other.getY();
}

// Conversion constructor from Number1 to Number2
Number2::Number2(const Number1& other) {
  x = other.getX();
  y = other.getY();
}

int main() {
  Number1 n1(1, 2);
  Number2 n2(3, 4);

  Number1 n3 = n1 + n2;
  Number2 n4 = n1 + n2;

  n1.print(); // Prints "1 2"
  n2.print(); // Prints "3 4"
  n3.print(); // Prints "4 6" (Number1 + Number2)
  n4.print(); // Prints "4 6" (Number2 + Number1)

  return 0;
}

#include <iostream>
using namespace std;
const int size =5;

class Number{
  int arr[size];
  
  public:
    Number(){
       for(int i=0;i<=size;i++)
         arr[i] = i;
    }
    int operator[](int i){
        return arr[i];
    }
};    

int main()
{
    Number n1;
  
    cout<<n1[5];

    return 0;
}

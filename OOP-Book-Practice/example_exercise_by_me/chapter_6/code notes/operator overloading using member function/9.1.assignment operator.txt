#include <iostream>
using namespace std;

class Number{
  int x,y;
  
  public:
    Number(){
        x=0;
        y=0;
    }
    Number(int a,int b){
        x=a;
        y=b;
    }
    void print(){
        cout<<x<<" "<<y<<endl;
    }
    
    void operator=(Number ob){
        x=ob.x;
        y=ob.y;
    }
};    

int main()
{
    Number n1,n2(5,6),n3;
    n1=n2;
    n1.print();
    n2.print();
    return 0;
}

#include <iostream>
using namespace std;

class Number {
  int x, y;

public:
  Number() {
    x = 0;
    y = 0;
  }

  Number(int a, int b) {
    x = a;
    y = b;
  }

  void print() {
    cout << x << " " << y;
  }

  // Overload the << operator using a member function
  void operator<<(ostream& os) {
    os << x << " " << y;
  }
};

int main() {
  Number n1(5, 6);

  cout << "Using print method: ";
  n1.print();
  cout << endl;

  cout << "Using overloaded << operator: ";
  n1 << cout;
  cout << endl;

  return 0;
}

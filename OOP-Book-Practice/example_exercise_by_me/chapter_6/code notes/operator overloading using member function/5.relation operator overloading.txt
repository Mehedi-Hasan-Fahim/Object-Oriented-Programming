#include <iostream>
using namespace std;

class Number{
  int x,y;
  
  public:
    Number(){
        x=0;
        y=0;
    }
    Number(int a,int b){
        x=a;
        y=b;
    }
    void print(){
        cout<<x<<" "<<y<<endl;
    }
    
   
    int operator > (Number ob){
        return (x+y) >(ob.x + ob.y) ;
    }
    
    int operator < (Number ob){
        return (x+y) < (ob.x + ob.y) ;
    }
    
    int operator == (Number ob){
        return (x+y) == (ob.x + ob.y) ;
    }

};    


int main()
{
    Number n1(6,6),n2(8,6),n3;
    
    if(n1>n2) cout<<"N1 is greater than N2"<<endl;
    else if(n1==n2) cout<<"N1 is equal to N2"<<endl;
    else cout<<"N2 is greater than N1"<<endl;
  
    return 0;
}

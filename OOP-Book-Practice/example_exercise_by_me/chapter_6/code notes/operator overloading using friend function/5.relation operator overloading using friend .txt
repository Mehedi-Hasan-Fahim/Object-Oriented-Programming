#include <iostream>
using namespace std;

class Number{
  int x,y;
  
  public:
    Number(){
        x=0;
        y=0;
    }
    Number(int a,int b){
        x=a;
        y=b;
    }
    void print(){
        cout<<x<<" "<<y<<endl;
    }
    
   
    friend int operator > (Number ob1,Number ob2){
        return (ob1.x+ob1.y) >(ob2.x + ob2.y) ;
    }
    
    friend int operator < (Number ob1,Number ob2){
        return (ob1.x+ob1.y) < (ob2.x + ob2.y) ;
    }
    
   friend int operator == (Number ob1,Number ob2){
         return (ob1.x+ob1.y) == (ob2.x + ob2.y) ;
    }

};    


int main()
{
    Number n1(6,6),n2(8,6),n3;
    
    if(n1>n2) cout<<"N1 is greater than N2"<<endl;
    else if(n1==n2) cout<<"N1 is equal to N2"<<endl;
    else cout<<"N2 is greater than N1"<<endl;
  
    return 0;
}

#include <iostream>
using namespace std;

class Number{
  int x,y;
  
  public:
    Number(){
        x=0;
        y=0;
    }
    Number(int a,int b){
        x=a;
        y=b;
    }
    void print(){
        cout<<x<<" "<<y<<endl;
    }
   
   friend void operator << (ostream& ob1,Number ob2){
      ob1<<ob2.x<<" "<<ob2.y<<endl;
    } 
    
}; 

int main()
{
    Number n1(10,6);
    
    cout<<n1;
    
    return 0;
}

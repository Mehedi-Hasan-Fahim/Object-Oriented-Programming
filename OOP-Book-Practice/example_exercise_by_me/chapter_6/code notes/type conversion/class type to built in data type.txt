#include <iostream>
using namespace std;

class Number{
  int x,y;
  public:
  Number(){
      x=0;
      y=0;
  }
   Number(int a,int b){
      x=a;
      y=b;
  }
   Number(int a){
      x=a;
      y=0;
  }
  void print(){
      cout<<x<<" "<<y<<endl;
  }
  
  operator int(){
      return x+y;
  }
    
};

int main()
{
    Number n1(2,4);
    int a;
    
    //class to built in data type
    
    a=n1;
    cout<<a;//output 2+4=6
    
    return 0;
}

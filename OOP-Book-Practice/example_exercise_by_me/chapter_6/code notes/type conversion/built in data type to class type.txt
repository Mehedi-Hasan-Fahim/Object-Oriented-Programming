#include <iostream>
using namespace std;

class Number{
  int x,y;
  public:
  Number(){
      x=0;
      y=0;
  }
   Number(int a,int b){
      x=a;
      y=b;
  }
   Number(int a){
      x=a;
      y=0;
  }
  void print(){
      cout<<x<<" "<<y<<endl;
  }
     
};

int main()
{
    Number n1(2,4);
    int a=5;
    
    // built in data type to class
    
    n1 =a;
    n1.print();
    
    return 0;
}

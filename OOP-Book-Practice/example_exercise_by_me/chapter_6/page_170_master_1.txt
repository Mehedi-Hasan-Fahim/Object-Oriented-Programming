#include <iostream>
using namespace std;

class coord{
  int x,y;
  public:
  coord(){
      x=0;y=0;
  }
  coord(int i,int j){
      x=i;y=j;
  }
  
  void get_xy(){
      cout<<x<<" "<<y<<endl;
  }
  
  coord operator <<(int i){
      coord tmp;
      tmp.x = x << i ;
      tmp.y = y << i ;
      
      return tmp;
  }
  coord operator >>(int i){
      coord tmp;
      tmp.x = x >> i ;
      tmp.y = y >> i ;
      
      return tmp;
  }
    
};


int main()
{
    coord o1(4,6),o2;
    
    o2 = o1 << 2;
     o2.get_xy();
     o2 = o1 >> 2;
     o2.get_xy();
    

    return 0;
}

#include<iostream>
#include<ctime>
using namespace std;
class stopwatch{
    clock_t begin,end;
    public:
       stopwatch();
       ~stopwatch();
       void start();
       void stop();
       void show();
};
stopwatch::stopwatch(){
    begin=end=0;
}
stopwatch::~stopwatch(){
    cout<<"Stopwatch object is destroyed"<<endl;
    show();
}
void stopwatch::start(){
    begin=clock()/CLOCKS_PER_SEC ;
}
void stopwatch::stop(){
    end=clock()/CLOCKS_PER_SEC ;
}
void stopwatch::show(){
    cout<<"Elapsed time "<<end-begin<<endl;
}
int main(){
    stopwatch ob;
    int ch;
    cout<<"Press 1 to Start:"<<endl;

    switch(ch){

               cin>>i;

                     case 2:
                           ob.stop();
                           break;
                        }

      break;
    }




}
